    function updateTaluks() {
      const districtSelect = document.getElementById("district");
      const talukSelect = document.getElementById("taluk");
      const selectedDistrict = districtSelect.value;

      talukSelect.innerHTML = '<option value="all">அனைத்து வட்டங்கள்</option>';

      if (selectedDistrict !== "all" && talukData[selectedDistrict]) {
        talukData[selectedDistrict].forEach(taluk => {
          const option = document.createElement("option");
          option.value = taluk;
          option.textContent = taluk;
          talukSelect.appendChild(option);
        });
      }
    }

    function filterData() {
      const district = document.getElementById("district").value;
      const taluk = document.getElementById("taluk").value;
      const status = document.getElementById("status").value;
      const search = document.getElementById("search").value;
      
      alert(`தேடல் விவரங்கள்:\nமாவட்டம்: ${district}\nதாலுகா: ${taluk}\nநிலை: ${status}\nகீওয়ার্ড: ${search}`);
    }
